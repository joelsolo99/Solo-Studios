<script>
  document.addEventListener("DOMContentLoaded", function () {

    const faders = document.querySelectorAll(
      ".package-intro-section, .package-options-section, .package-advice-section, .packages-cta"
    );


    const appearOptions = {
      threshold: 0.1,
      rootMargin: "0px 0px -100px 0px"
    };


    const appearOnScroll = new IntersectionObserver(function (entries, observer) {

      entries.forEach(function (entry) {

        if (!entry.isIntersecting) return;

        entry.target.classList.add("visible");

        observer.unobserve(entry.target);

      });


    }, appearOptions);



    faders.forEach(function (fader) {

      appearOnScroll.observe(fader);

    });



    if (typeof snowStorm !== "undefined") {

      snowStorm.snowColor = "#38b7c2";
      snowStorm.flakesMax = 220;
      snowStorm.animationInterval = 40;
      snowStorm.flakeWidth = 14;
      snowStorm.flakeHeight = 14;

    }


  });
</script>
